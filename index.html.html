ภาพตารางและการดึงข้อมูล

กด ถัดไป เพื่อดูว่าข้อมูลเปลี่ยนไปอย่างไรในแต่ละขั้นตอน

SELECT d.name, COUNT(*) AS total, AVG(e.salary) AS avg_sal
FROM employees e
JOIN departments d ON e.dept_id = d.id
WHERE e.salary > 50000
GROUP BY d.name
HAVING COUNT(*) >= 1
ORDER BY avg_sal DESC
LIMIT 2
